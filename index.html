Run this SQL in your Supabase SQL editor to enable workout tracking:
create table public.workout_logs (
  id bigint generated always as identity primary key,
  date date not null,
  exercise text not null,
  sets integer,
  reps integer,
  weight_kg numeric,
  notes text,
  created_at timestamptz default now()
);
alter table public.workout_logs enable row level security;
create policy "anon_select" on public.workout_logs for select using (true);
create policy "anon_insert" on public.workout_logs for insert with check (true);
create policy "anon_update" on public.workout_logs for update using (true);
create policy "anon_delete" on public.workout_logs for delete using (true);
